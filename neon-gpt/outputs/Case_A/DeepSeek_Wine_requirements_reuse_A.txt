:WineOntology a owl:Ontology ;
    rdfs:label "Wine Ontology"@en ;
    rdfs:comment "An ontology for representing knowledge in the domain of viticulture and oenology."@en ;

# Core Classes
:Wine a owl:Class ;
    rdfs:label "Wine"@en ;
    rdfs:comment "A beverage produced by the fermentation of grape juice."@en ;
    rdfs:subClassOf foodon:03301106 ; # Aligns with FoodOn 'wine' class
    owl:disjointWith :GrapeVariety, :Winery, :Vintage .

:GrapeVariety a owl:Class ;
    rdfs:label "Grape Variety"@en ;

    rdfs:subClassOf foodon:00002403 ; # Aligns with FoodOn 'plant food product'

:WineRegion a owl:Class ;

:Winery a owl:Class ;

:Vintage a owl:Class ;

:TastingNote a owl:Class ;
    rdfs:label "Tasting Note"@en ;
